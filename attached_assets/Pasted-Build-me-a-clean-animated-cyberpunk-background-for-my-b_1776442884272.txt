  <script>
    const PALETTE = {
      building: ["#111215", "#17181c", "#1d1f24"],
      neon: ["#ff4fa3", "#ff7a2f", "#ff4d4d", "#39d5ff"]
    };

    const reduceMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;

    function rand(min, max) {
      return Math.random() * (max - min) + min;
    }

    function randInt(min, max) {
      return Math.floor(rand(min, max + 1));
    }

    function pick(arr) {
      return arr[Math.floor(Math.random() * arr.length)];
    }

    function createSegment(config) {
      const segment = document.createElement("div");
      segment.className = "segment";

      const width = window.innerWidth || 1400;
      let x = -40;

      while (x < width + 120) {
        const b = document.createElement("div");
        b.className = "building";

        const w = randInt(config.widthMin, config.widthMax);
        const h = randInt(config.heightMin, config.heightMax);

        b.style.left = `${x}px`;
        b.style.width = `${w}px`;
        b.style.height = `${h}px`;
        b.style.background = pick(PALETTE.building);

        if (Math.random() < config.roofChance) {
          const roof = document.createElement("div");
          roof.className = "roof-detail";
          roof.style.width = `${randInt(2, 4)}px`;
          roof.style.height = `${randInt(10, 24)}px`;
          roof.style.left = `${randInt(6, Math.max(6, w - 10))}px`;
          roof.style.bottom = `${h - 2}px`;
          segment.appendChild(roof);
        }

        const cols = Math.max(2, Math.floor(w / randInt(14, 18)));
        const rows = Math.max(2, Math.floor(h / randInt(18, 24)));
        const padX = randInt(6, 10);
        const padY = randInt(8, 12);
        const gapX = randInt(6, 10);
        const gapY = randInt(7, 11);
        const ww = Math.max(3, Math.floor((w - padX * 2 - gapX * (cols - 1)) / cols));
        const wh = randInt(4, 8);

        for (let row = 0; row < rows; row++) {
          for (let col = 0; col < cols; col++) {
            const win = document.createElement("div");
            win.className = "window";

            const lit = Math.random() < config.lightChance;
            const color = lit ? pick(PALETTE.neon) : "rgba(255,255,255,0.04)";

            win.style.width = `${ww}px`;
            win.style.height = `${wh}px`;
            win.style.left = `${padX + col * (ww + gapX)}px`;
            win.style.top = `${padY + row * (wh + gapY)}px`;
            win.style.background = color;

            if (lit) {
              const blur = color === "#39d5ff" ? 7 : 6;
              win.style.boxShadow = `0 0 ${blur}px ${color}`;
              win.dataset.lit = "1";
              win.dataset.base = color;
            } else {
              win.classList.add("off");
              win.dataset.lit = "0";
            }

            b.appendChild(win);
          }
        }

        segment.appendChild(b);
        x += w + randInt(config.gapMin, config.gapMax);
      }

      if (config.lampCount > 0) {
        for (let i = 0; i < config.lampCount; i++) {
          const lamp = document.createElement("div");
          lamp.className = "lamp";
          lamp.style.left = `${randInt(0, width)}px`;
          lamp.style.height = `${randInt(16, 34)}px`;
          lamp.style.transform = `rotate(${rand(-8, 8)}deg)`;

          const head = document.createElement("div");
          head.className = "lamp-head";
          const glow = pick(["#ff4fa3", "#ff7a2f", "#ff4d4d"]);
          head.style.background = glow;
          head.style.boxShadow = `0 0 10px ${glow}, 0 0 18px ${glow}`;
          lamp.appendChild(head);

          segment.appendChild(lamp);
        }
      }

      return segment;
    }

    function fillTrack(trackId, config) {
      const track = document.getElementById(trackId);
      track.innerHTML = "";
      track.appendChild(createSegment(config));
      track.appendChild(createSegment(config));
    }

    function buildScene() {
      fillTrack("backTrack", {
        widthMin: 70,
        widthMax: 150,
        heightMin: 60,
        heightMax: 130,
        gapMin: 8,
        gapMax: 22,
        lightChance: 0.04,
        roofChance: 0.12,
        lampCount: 0
      });

      fillTrack("midTrack", {
        widthMin: 80,
        widthMax: 180,
        heightMin: 80,
        heightMax: 170,
        gapMin: 10,
        gapMax: 24,
        lightChance: 0.08,
        roofChance: 0.16,
        lampCount: 2
      });

      fillTrack("frontTrack", {
        widthMin: 90,
        widthMax: 210,
        heightMin: 90,
        heightMax: 190,
        gapMin: 12,
        gapMax: 26,
        lightChance: 0.11,
        roofChance: 0.2,
        lampCount: 5
      });
    }

    function twinkleWindows() {
      if (reduceMotion) return;

      const litWindows = [...document.querySelectorAll('.window[data-lit="1"]')];
      if (!litWindows.length) return;

      setInterval(() => {
        const batch = litWindows.sort(() => 0.5 - Math.random()).slice(0, Math.max(4, Math.floor(litWindows.length * 0.015)));

        batch.forEach((win, i) => {
          const base = win.dataset.base || "#ff7a2f";
          const shouldDim = Math.random() < 0.65;

          setTimeout(() => {
            if (shouldDim) {
              win.style.opacity = "0.35";
              win.style.filter = "blur(0.3px)";
              win.style.boxShadow = "none";
            } else {
              win.style.opacity = "1";
              win.style.filter = "none";
              win.style.boxShadow = `0 0 8px ${base}`;
            }

            setTimeout(() => {
              win.style.opacity = "0.9";
              win.style.filter = "none";
              win.style.boxShadow = `0 0 6px ${base}`;
            }, 1800);
          }, i * 120);
        });
      }, 5000);
    }

    buildScene();
    twinkleWindows();

    let resizeTimer;
    window.addEventListener("resize", () => {
      clearTimeout(resizeTimer);
      resizeTimer = setTimeout(() => {
        buildScene();
      }, 180);
    });
  </script>
